cmake_minimum_required(VERSION 3.8)
project(office4_route)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
add_library(route_tool SHARED src/route_tool.cpp)
target_compile_features(route_tool PUBLIC cxx_std_17)
ament_target_dependencies(route_tool rviz_common rviz_rendering pluginlib nav_msgs rclcpp)
pluginlib_export_plugin_description_file(rviz_common plugin.xml)
find_package(nav2_core REQUIRED)
find_package(nav2_controller REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(route_controller SHARED src/route_controller.cpp)
target_compile_features(route_controller PUBLIC cxx_std_17)
ament_target_dependencies(route_controller nav2_controller nav2_core nav2_regulated_pure_pursuit_controller
  pluginlib visualization_msgs std_msgs)
pluginlib_export_plugin_description_file(nav2_core controller_plugin.xml)
install(TARGETS route_controller route_tool LIBRARY DESTINATION lib)
install(DIRECTORY config scripts DESTINATION share/${PROJECT_NAME})
ament_package()
